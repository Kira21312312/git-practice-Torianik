git log - информация о комитах их история
git init -b main - создать локальный репозиторий и сразу сменить имя ветки на main
touch R.txt - создать файл
git remote add origin git@github.../https:.... - локальный репозиторий соединить с гитхабовским
git add . - добавить все файлы новые или измененые что бы учитывались
git commit -m "dhfja" - создать комит -m дает возможность не открывать текстовый редактор что бы дать описание
git push -u origin - достаточно писать один раз связывает ветку main с origin потом можно писать просто git push\pull
git switch -c ... - создать ветку и переключится на нее
git switch ... переключится на ветку
git switch .. - переключится на прошлую ветку
git log --oneline -инфа о последнем комите
git branch -M main - переменовать текущую ветку
